decl
    int factorial(int n,int n);
enddecl

int factorial(int n,int n){
    decl
        int f;
    enddecl
    begin
        if( (n==1) OR (n==0) ) then
            f = 1;
        else
            f = n * factorial(n-1);
        endif;
        return f;
    end;
}

int main(){
    decl
        int a;
    enddecl
    begin
        read(a);
        factorial = factorial(a,2);
        write(factorial);
        return 0;
    end;
}
