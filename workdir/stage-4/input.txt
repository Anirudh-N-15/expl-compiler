decl
    int A[10][10], B[10][10], C[10][10];
    int n, m, p, q;
    int i, j, k;
enddecl

begin
    read(n);
    read(m);
    read(p);
    read(q);

    if (m != p) then
        write("Error"); 
        Exit;
    endif;

    i = 0;
    while(i < n) do
        j = 0;
        while(j < m) do
            read(A[i][j]);
            j = j + 1;
        endwhile;
        i = i + 1;
    endwhile;

    i = 0;
    while(i < p) do
        j = 0;
        while(j < q) do
            read(B[i][j]);
            j = j + 1;
        endwhile;
        i = i + 1;
    endwhile;

    i = 0;
    while(i < n) do
        j = 0;
        while(j < q) do
            C[i][j] = 0;
            j = j + 1;
        endwhile;
        i = i + 1;
    endwhile;

    i = 0;
    while(i < n) do
        j = 0;
        while(j < q) do
            k = 0;
            while(k < m) do
                C[i][j] = C[i][j] + (A[i][k] * B[k][j]);
                k = k + 1;
            endwhile;
            j = j + 1;
        endwhile;
        i = i + 1;
    endwhile;

    i = 0;
    while(i < n) do
        j = 0;
        while(j < q) do
            write(C[i][j]);
            j = j + 1;
        endwhile;
        i = i + 1;
    endwhile;
end;
